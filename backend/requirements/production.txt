# AEGIS Backend - Production Requirements
-r base.txt

# WSGI server for the Render web service
# (gunicorn config.wsgi:application --bind 0.0.0.0:$PORT).
gunicorn>=26,<27

# Shared cache for the rate limits.
#
# The DRF throttles (login, refresh, import, export, password change,
# registration) run through the default cache. The local-memory cache is
# per-process, so with more than one gunicorn worker every limit is
# multiplied by the number of workers. Point REDIS_URL at a Redis instance
# and these limits hold across the whole deployment.
django-redis>=5,<7

# Uploaded media (profile photos) in Supabase Storage.
#
# The free Render plan has no persistent disk, so files written to the
# service's disk are lost on the next redeploy. With these two packages and
# the SUPABASE_S3_* variables set, uploads go to a Supabase bucket instead.
# Without the variables the app keeps using the local disk, so nothing
# changes for a deployment that has not configured storage.
django-storages>=1.14,<2.0
boto3>=1.35,<2.0
